# .gitlab-ci.yml — 直接拿去用 stages: - quality-gate - ai-review - notify sonarqube-check: stage: quality-gate image: sonarsource/sonar-scanner-cli:latest only: [merge_requests] script: - sonar-scanner \ -Dsonar.host.url=$SONAR_HOST_URL \ -Dsonar.login=$SONAR_TOKEN \ -Dsonar.qualitygate.wait=true ai-code-review: stage: ai-review image: alpine/curl:latest only: [merge_requests] needs: [sonarqube-check] script: - MR_DIFF=$(curl -s --header "PRIVATE-TOKEN: ${GITLAB_ACCESS_TOKEN}" \ "${CI_API_V4_URL}/projects/${CI_PROJECT_ID}/merge_requests/${CI_MERGE_REQUEST_IID}/changes" \ | jq -r '.changes[] | "File: \(.old_path)\n\(.diff)"') - REVIEW=$(curl -s -X POST https://lewislunora.onrender.com/api/ai-review \ -H "Authorization: Bearer ${AI_REVIEW_KEY}" \ -H "Content-Type: application/json" \ -d "$(jq -n --arg diff "$MR_DIFF" '{diff: $diff}')") - curl -s --request POST --header "PRIVATE-TOKEN: ${GITLAB_ACCESS_TOKEN}" \ "${CI_API_V4_URL}/projects/${CI_PROJECT_ID}/merge_requests/${CI_MERGE_REQUEST_IID}/notes" \ --data "body=🤖 **AI Review**\n${REVIEW}"
